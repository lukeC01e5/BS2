#include <Keypad.h>
#include <TFT_eSPI.h> // TFT display library
#include <Arduino.h>
#include <SPI.h>
#include <FS.h>
#include <MFRC522.h>
#include "RFIDData.h"
#include "GlobalDefs.h" // Ensure this defines SCK_PIN, MISO_PIN, MOSI_PIN, SS_PIN, and a global 'key' of type MFRC522::MIFARE_Key

// Keypad configuration (4x3 keypad)
const byte ROWS = 4;
const byte COLS = 3;
char keys[ROWS][COLS] = {
    {'1', '2', '3'},
    {'4', '5', '6'},
    {'7', '8', '9'},
    {'*', '0', '#'}};
byte rowPins[ROWS] = {21, 17, 2, 15};
byte colPins[COLS] = {12, 27, 32};

Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);
TFT_eSPI tft; // TFT display instance

// Function to get challenge code from keypad
String getChallengeCodeFromKeypad()
{
    String code = "";
    tft.fillScreen(TFT_BLACK);
    tft.setCursor(0, 0);
    tft.println("Enter 3-digit code:");
    while (code.length() < 3)
    {
        char key = keypad.getKey();
        if (key && isDigit(key))
        {
            code += key;
            tft.print(key);
        }
    }
    return code;
}

void setup()
{
    Serial.begin(115200);
    tft.init();
    tft.setRotation(3);
    tft.setTextSize(2);
    tft.setTextColor(TFT_WHITE, TFT_BLACK);
    tft.fillScreen(TFT_BLACK);
    tft.setCursor(0, 0);
    tft.println("TFT Initialized\n");

    Serial.println("About to call SPI.begin()");
    // Use explicit pins for SPI and include SS_PIN (this was the key difference)
    SPI.begin(SCK_PIN, MISO_PIN, MOSI_PIN, SS_PIN);

    Serial.println("About to call mfrc522.PCD_Init()");
    mfrc522.PCD_Init();
    Serial.println("Done initializing RFID");

    // Initialize global default key if not already done (set all 6 bytes to 0xFF)
    for (byte i = 0; i < 6; i++)
    {
        key.keyByte[i] = 0xFF;
    }

    // Welcome and wait for a key press before proceeding
    tft.fillScreen(TFT_BLACK);
    tft.setCursor(0, 0);
    tft.setTextColor(TFT_GREEN, TFT_BLACK);
    tft.println("press any key\n to start");
    while (keypad.getKey() == NO_KEY)
    {
        delay(100);
    }
    tft.fillScreen(TFT_BLACK);
}

void loop()
{
    // 1) Get the 3-digit challenge code from the user
    String newChallengeCode = getChallengeCodeFromKeypad(); // e.g., "111"
    tft.println("\nGot: " + newChallengeCode);

    // 2) Prompt for RFID tag and wait until one is present
    tft.println("\nPresent RFID Tag...");
    while (!mfrc522.PICC_IsNewCardPresent() || !mfrc522.PICC_ReadCardSerial())
    {
        delay(100);
    }
    tft.println("Tag present!");
    delay(200);

    // Before waking up the card, declare a buffer for ATQA
    byte bufferATQA[2];
    byte bufferSize = sizeof(bufferATQA);

    // Wake up card (you only need one wakeup call)
    mfrc522.PICC_WakeupA(bufferATQA, &bufferSize);
    delay(200);

    tft.println("Testing basic card functions...");

    // Check that the card responds and print its UID
    if (mfrc522.PICC_IsNewCardPresent() && mfrc522.PICC_ReadCardSerial())
    {
        tft.println("Card UID: ");
        String uidString = "";
        for (byte i = 0; i < mfrc522.uid.size; i++)
        {
            uidString += String(mfrc522.uid.uidByte[i] < 0x10 ? "0" : "");
            uidString += String(mfrc522.uid.uidByte[i], HEX);
        }
        tft.println(uidString);

        // Attempt authentication using the default key
        MFRC522::MIFARE_Key testKey;
        for (byte i = 0; i < 6; i++)
        {
            testKey.keyByte[i] = 0xFF;
        }
        byte testBlock = 1;
        byte trailerBlock = (testBlock / 4) * 4 + 3;
        MFRC522::StatusCode status = mfrc522.PCD_Authenticate(
            MFRC522::PICC_CMD_MF_AUTH_KEY_A, trailerBlock, &testKey, &(mfrc522.uid));
        if (status == MFRC522::STATUS_OK)
        {
            tft.println("Auth OK!");
        }
        else
        {
            tft.println("Auth failed: ");
            tft.println(mfrc522.GetStatusCodeName(status));
        }
    }

    // Re-use the earlier buffer and wakeup procedure before reading
    bufferSize = sizeof(bufferATQA);
    mfrc522.PICC_WakeupA(bufferATQA, &bufferSize);
    delay(200);

    // 3) Read existing data from block 1 using your RFIDData read function
    int myIntPart = 0;
    String myStrPart;
    String rawData = readFromRFID(mfrc522, key, 1, myIntPart, myStrPart);

    // Debug output on TFT and Serial
    tft.fillScreen(TFT_BLACK);
    tft.setCursor(0, 0);
    tft.println("Raw data read:");
    tft.println(rawData);
    tft.println("Length: " + String(rawData.length()));
    tft.println("Has %: " + String(rawData.indexOf('%') != -1 ? "YES" : "NO"));
    Serial.println("DEBUG - Raw data: " + rawData);
    Serial.println("DEBUG - Length: " + String(rawData.length()));
    Serial.println("DEBUG - Has %: " + String(rawData.indexOf('%') != -1 ? "YES" : "NO"));

    delay(3000); // Allow time to review debug output

    if (rawData.indexOf('%') != -1 && rawData.length() >= 7)
    {
        // Parse the existing data
        String wrong = rawData.substring(3, 4); // e.g., existing wrong guess digit
        String bools = rawData.substring(5, 7); // e.g., existing bool value
        int pctPos = rawData.indexOf('%');
        String namePart = rawData.substring(pctPos + 1);

        // Build the updated RFID string (replace first 3 challenge digits)
        String updatedString = newChallengeCode + wrong + "?" + bools + "%" + namePart;
        delay(200);

        // Write the updated string to the card using the RFIDData write function
        MFRC522::MIFARE_Key mainKey;
        for (byte i = 0; i < 6; i++)
        {
            mainKey.keyByte[i] = 0xFF;
        }
        bool writeOk = writeToRFID(mfrc522, mainKey, updatedString, 1);
        if (writeOk)
        {
            Serial.println("RFID update succeeded!");
            tft.println("Update Succeeded!");
        }
        else
        {
            Serial.println("RFID update failed!");
            tft.println("Update Failed!");
        }
    }
    else
    {
        tft.println("Invalid existing data format.");
    }

    // Halt card and stop crypto session
    mfrc522.PICC_HaltA();
    mfrc522.PCD_StopCrypto1();

    delay(2000); // Pause before restarting loop
}
