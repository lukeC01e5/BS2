#include <TFT_eSPI.h>         // TFT display library
#include <Arduino.h>
#include <SPI.h>
#include <MFRC522.h>
#include "RFIDData.h"         // Contains readFromRFID()
#include "GlobalDefs.h"       // Defines SCK_PIN, MISO_PIN, MOSI_PIN, SS_PIN, RST_PIN, and global key

TFT_eSPI tft;                 // TFT display instance
// MFRC522 mfrc522(SS_PIN, RST_PIN);

void setup() {
    Serial.begin(115200);
    tft.init();
    tft.setRotation(3);
    tft.setTextSize(2);
    tft.setTextColor(TFT_WHITE, TFT_BLACK);
    tft.fillScreen(TFT_BLACK);
    tft.setCursor(0, 0);
    tft.println("RFID Tag Reader");

    Serial.println("Initializing SPI...");
    SPI.begin(SCK_PIN, MISO_PIN, MOSI_PIN, SS_PIN);
    Serial.println("Initializing RFID...");
    mfrc522.PCD_Init();
    Serial.println("RFID Initialized");

    // Set global default key (all 6 bytes = 0xFF)
    for (byte i = 0; i < 6; i++) {
        key.keyByte[i] = 0xFF;
    }
}

void loop() {
    // Wait for a new RFID card
    if (mfrc522.PICC_IsNewCardPresent() && mfrc522.PICC_ReadCardSerial()) {

        // Read raw data from block 1 using your RFIDData function.
        int dummyInt = 0;
        String dummyStr;
        String rawData = readFromRFID(mfrc522, key, 1, dummyInt, dummyStr);

        // Display raw data on TFT
        tft.fillScreen(TFT_BLACK);
        tft.setCursor(0, 0);
        tft.println("Raw RFID Data:");
        tft.println(rawData);

        // Also output to Serial
        Serial.println("Raw RFID Data:");
        Serial.println(rawData);

        // Halt the card and stop encryption
        mfrc522.PICC_HaltA();
        mfrc522.PCD_StopCrypto1();

        delay(2000);  // Pause before reading next card
    }
}
